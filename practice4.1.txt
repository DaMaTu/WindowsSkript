# Prompt the user to enter two values
$value1 = Read-Host "Enter the first value"
$value2 = Read-Host "Enter the second value"

# Compare the two values
if ($value1 -gt $value2) {
    Write-Host "The higher number is : $value1"  # If first value is greater
} else {
    Write-Host "The higher number is : $value2"  # If second value is greater or equal
}
